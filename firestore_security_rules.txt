// Firebase Firestore Security Rules for Posts Collection
// Copy these rules to your Firebase Console -> Firestore Database -> Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Posts collection rules
    match /posts/{postId} {
      
      // Allow read access to all authenticated users
      allow read: if request.auth != null;
      
      // Allow create if:
      // - User is authenticated
      // - User is the author of the post
      // - Author ID matches the current user's UID
      // - Required fields are present and valid
      allow create: if request.auth != null 
        && request.auth.uid == resource.data.authorId
        && isValidPostData(request.resource.data)
        && request.resource.data.authorId == request.auth.uid;
      
      // Allow update if:
      // - User is authenticated
      // - User is the author of the post
      // - Core fields cannot be changed (authorId, createdAt)
      // - Updated data is valid
      allow update: if request.auth != null 
        && request.auth.uid == resource.data.authorId
        && resource.data.authorId == request.resource.data.authorId
        && resource.data.createdAt == request.resource.data.createdAt
        && isValidPostData(request.resource.data);
      
      // Allow delete only by the post author
      allow delete: if request.auth != null 
        && request.auth.uid == resource.data.authorId;
    }
    
    // Helper function to validate post data
    function isValidPostData(data) {
      return data.keys().hasAll(['id', 'title', 'description', 'authorId', 'authorName', 'createdAt', 'status', 'participantIds', 'maxParticipants', 'genderPreferences'])
        && data.title is string
        && data.title.size() > 0
        && data.title.size() <= 100
        && data.description is string
        && data.description.size() <= 500
        && data.authorId is string
        && data.authorName is string
        && data.createdAt is timestamp
        && data.status in ['upcoming', 'ongoing', 'completed']
        && data.participantIds is list
        && data.participantIds.size() <= data.maxParticipants
        && data.maxParticipants is number
        && data.maxParticipants > 0
        && data.maxParticipants <= 50
        && data.genderPreferences is list
        && data.genderPreferences.size() > 0
        && isValidGenderPreferences(data.genderPreferences)
        && (data.scheduledTime == null || data.scheduledTime is timestamp)
        && (data.location == null || (data.location is string && data.location.size() <= 200));
    }
    
    // Helper function to validate gender preferences
    function isValidGenderPreferences(preferences) {
      return preferences.toSet().difference(['Anyone', 'Male', 'Female', 'Non-binary'].toSet()).size() == 0;
    }
    
    // Additional rules for other collections if needed
    // Users collection (if you have one)
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;

      // Allow create for new users
      allow create: if request.auth != null && request.auth.uid == userId;

      // Allow update with gender change validation
      allow update: if request.auth != null
        && request.auth.uid == userId
        && validateGenderChange(resource.data, request.resource.data);
    }

    // Helper function to validate gender changes
    function validateGenderChange(oldData, newData) {
      // Whitelisted emails that can change gender unlimited times
      let whitelistedEmails = ['jordangr@bu.edu', 'enteigss@gmail.com'];
      let userEmail = newData.email.lower();
      let isWhitelisted = userEmail in whitelistedEmails;

      // Check if gender is being changed
      let isGenderChanging = 'gender' in oldData
        && 'gender' in newData
        && oldData.gender != null
        && oldData.gender != newData.gender;

      // If gender is not changing, allow the update
      if (!isGenderChanging) {
        return true;
      }

      // If gender is changing and user is whitelisted, allow it
      if (isWhitelisted) {
        return true;
      }

      // For non-whitelisted users, check they haven't exceeded limit
      let oldChangeCount = 'genderChangeCount' in oldData ? oldData.genderChangeCount : 0;
      let newChangeCount = 'genderChangeCount' in newData ? newData.genderChangeCount : 0;

      // Allow if they haven't used their one change yet and count is incremented correctly
      return oldChangeCount < 1 && newChangeCount == oldChangeCount + 1;
    }
    
    // Groups/Squads collection (if you have one)
    match /groups/{groupId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null 
        && (resource == null || request.auth.uid in resource.data.memberIds);
    }
  }
}

// Additional Notes:
// 1. These rules ensure that only authenticated users can interact with posts
// 2. Users can only create/edit/delete their own posts
// 3. Anyone can join/leave posts by updating the participantIds array
// 4. Data validation ensures posts have proper structure and limits
// 5. Gender preferences are validated against allowed values
// 6. Consider adding rate limiting in production
// 7. You may want to add more specific rules for joining/leaving posts

// To implement join/leave functionality securely, consider using Cloud Functions
// that can atomically update participant lists with proper validation.
